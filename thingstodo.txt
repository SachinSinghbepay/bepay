// --------------------------- temprorary file for diffing ---------------------------
import { initializeApp, getApps } from "firebase/app"
import {
  getFirestore,
  collection,
  addDoc,
  query,
  where,
  getDocs,
} from "firebase/firestore"
import { getAuth } from "firebase/auth"

// 🔹 Firebase config from .env.local
const firebaseConfig = {
  apiKey: process.env.NEXT_PUBLIC_FIREBASE_API_KEY,
  authDomain: process.env.NEXT_PUBLIC_FIREBASE_AUTH_DOMAIN,
  projectId: process.env.NEXT_PUBLIC_FIREBASE_PROJECT_ID,
  storageBucket: process.env.NEXT_PUBLIC_FIREBASE_STORAGE_BUCKET,
  messagingSenderId: process.env.NEXT_PUBLIC_FIREBASE_MESSAGING_SENDER_ID,
  appId: process.env.NEXT_PUBLIC_FIREBASE_APP_ID,
}

// ✅ Check if Firebase env exists
const hasFirebaseConfig =
  firebaseConfig.apiKey &&
  firebaseConfig.authDomain &&
  firebaseConfig.projectId

let app = null
let auth = null
let db = null

if (hasFirebaseConfig) {
  app = !getApps().length ? initializeApp(firebaseConfig) : getApps()[0]
  auth = getAuth(app)
  db = getFirestore(app)
} else {
  console.warn("⚠️ Firebase not initialized — missing env variables")
}

export { auth, db }

export const addToWaitlist = async (email, pathname = "/personal") => {
  try {
    // 🚫 If Firebase disabled, just fake success
    if (!db) {
      console.warn("Firebase disabled — skipping waitlist save")
      return { id: "local-dev", campaignId: "dev-mode" }
    }

    let collectionName = "waitlistPersonal"
    let source = "personal"

    if (pathname.includes("business")) {
      collectionName = "waitlistBusiness"
      source = "business"
    } else if (pathname.includes("upi")) {
      collectionName = "waitlistUpi"
      source = "upi"
    }

    const q = query(collection(db, collectionName), where("email", "==", email))
    const querySnapshot = await getDocs(q)

    if (!querySnapshot.empty) {
      throw new Error("Email already exists in this waitlist")
    }

    const newCampaignId =
      typeof window !== "undefined"
        ? localStorage.getItem("campaignId")
        : null

    const docRef = await addDoc(collection(db, collectionName), {
      email,
      campaignId: newCampaignId,
      timestamp: new Date(),
      source,
    })

    return { id: docRef.id, campaignId: newCampaignId }
  } catch (error) {
    throw error
  }
}
